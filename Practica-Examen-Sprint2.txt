        // GET: api/Modelos3D/GetModelos3DParaComprar
        [HttpGet]
        [Route("[action]")]
        [ProducesResponseType(typeof(IList<Modelo3DParaComprarDTO>), (int)HttpStatusCode.OK)]
        [ProducesResponseType(typeof(ValidationProblemDetails), (int)HttpStatusCode.BadRequest)]   // <- NUEVO
        public async Task<ActionResult> GetModelos3DParaComprar(string? nombre, string? categoria,
            FormatoModelo3D? formato, decimal? precioMaximo)                                       // <- NUEVOS PARÁMETROS
        {
            // NUEVO: validación del precio máximo
            if (precioMaximo != null && precioMaximo <= 0)
            {
                ModelState.AddModelError("precioMaximo", "Error! El precio máximo debe ser mayor que 0");
                return BadRequest(new ValidationProblemDetails(ModelState));
            }

            var modelosBD = await _context.Modelos3D
                .Include(m => m.Licencia)
                .Where(m => m.Licencia.FechaExpiracion >= DateTime.Today                // NUEVO: licencia no caducada
                         && (nombre == null || m.Nombre.Contains(nombre))
                         && (categoria == null || m.Categoria == categoria)
                         && (formato == null || m.Formato == formato))                    // NUEVO: filtro formato
                .ToListAsync();

            // El filtro y la ordenación por precio van DESPUÉS del ToListAsync:
            // SQLite (la BD de los tests) no sabe comparar ni ordenar decimal dentro de la consulta.
            IList<Modelo3DParaComprarDTO> modelos = modelosBD
                .Where(m => precioMaximo == null || m.Precio <= precioMaximo)            // NUEVO: precio máximo
                .OrderBy(m => m.Precio)                                                   // CAMBIADO: antes era Nombre
                .ThenBy(m => m.Nombre)                                                    // NUEVO
                .Select(m => new Modelo3DParaComprarDTO(m.Id, m.Nombre, m.Categoria,
                    m.Licencia.Nombre, m.Formato, m.Precio))
                .ToList();

            return Ok(modelos);
        }

        // GET: api/ComprasModelo3D/5
        [HttpGet]
        [Route("{id}")]
        [ProducesResponseType(typeof(CompraModelo3DDetalleDTO), (int)HttpStatusCode.OK)]
        [ProducesResponseType(typeof(ValidationProblemDetails), (int)HttpStatusCode.BadRequest)]   // <- NUEVO
        [ProducesResponseType((int)HttpStatusCode.NotFound)]
        public async Task<ActionResult> GetCompraModelo3D(int id)
        {
            // NUEVO: id inválido
            if (id <= 0)
            {
                ModelState.AddModelError("id", "Error! El identificador de la compra no es válido");
                return BadRequest(new ValidationProblemDetails(ModelState));
            }

            CompraModelo3D? compra = await _context.ComprasModelo3D
                .Include(c => c.Lineas)
                    .ThenInclude(l => l.Modelo)
                        .ThenInclude(m => m.Licencia)
                .FirstOrDefaultAsync(c => c.Id == id);

            if (compra == null)
            {
                _logger.LogError($"Error: la compra de modelos 3D con id {id} no existe");
                return NotFound();
            }

            var compraDTO = new CompraModelo3DDetalleDTO(compra.Id, compra.NombreCliente, compra.ApellidosCliente,
                compra.CorreoElectronico, compra.DireccionFacturacion, compra.FechaCompra, compra.PrecioTotal,
                compra.MetodoPago, compra.Descripcion,
                compra.Lineas
                    .OrderByDescending(l => l.Subtotal)                                            // <- NUEVO
                    .Select(l => new LineaCompraModeloDTO(l.Modelo.Id, l.Modelo.Nombre, l.Modelo.Categoria,
                        l.Modelo.Formato, l.Modelo.Licencia.Nombre, l.Modelo.Licencia.FechaExpiracion,
                        l.CantidadLicencias, l.PrecioUnidad, l.Subtotal,
                        l.Modelo.Licencia.FechaExpiracion < compra.FechaCompra.AddMonths(1)))      // <- NUEVO
                    .ToList());

            return Ok(compraDTO);
        }
